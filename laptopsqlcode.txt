-- Bảng CUSTOMERS
CREATE TABLE CUSTOMERS (
    customer_id SERIAL PRIMARY KEY,
    username VARCHAR(30) NOT NULL,
    email VARCHAR(100) NOT NULL,
    first_name VARCHAR(20) NOT NULL,
    last_name VARCHAR(20) NOT NULL,
    created_at TIMESTAMP,
    gender CHAR(1) CHECK (gender IN ('F', 'M')),
    address VARCHAR(100),
    date_of_birth DATE,
    phone VARCHAR(12)
);

-- Bảng PRODUCTS
CREATE TABLE PRODUCTS (
    product_id SERIAL PRIMARY KEY,
    product_name VARCHAR(20) NOT NULL,
    model VARCHAR(20) NOT NULL,
    brand VARCHAR(20) NOT NULL,
    description TEXT,
    price MONEY NOT NULL,
    stock_quantity INT NOT NULL,
    year_publish TIMESTAMP NOT NULL
);

-- Bảng category
CREATE TABLE categories(
    category_id SERIAL PRIMARY KEY,
    category_name VARCHAR(50) NOT NULL
);


-- Bảng EMPLOYEES
CREATE TABLE EMPLOYEES (
    employee_id SERIAL PRIMARY KEY,
    first_name VARCHAR(20),
    last_name VARCHAR(20),
    phone VARCHAR(12),
    address VARCHAR(100),
    gender CHAR(1) CHECK (gender IN ('F', 'M')),
    bank_number VARCHAR(10),
    role CHAR(10) NOT NULL,
    salary MONEY NOT NULL,
    work_day VARCHAR(200) NOT NULL,
    hire_day DATE NOT NULL
);

-- Bảng PAYMENTS
CREATE TABLE PAYMENTS (
    payment_id SERIAL PRIMARY KEY,
    employee_id INTEGER NOT NULL,
    payment_date TIMESTAMP NOT NULL,
    payment_method VARCHAR(20) NOT NULL,
    total_amount MONEY NOT NULL,
    status VARCHAR(15) NOT NULL,
    CONSTRAINT fk_employee_payment FOREIGN KEY (employee_id)
        REFERENCES EMPLOYEES (employee_id)
);

-- Bảng ORDERS
CREATE TABLE ORDERS (
    order_id SERIAL PRIMARY KEY,
    customer_id INTEGER NOT NULL,
    payment_id INTEGER NOT NULL,
    order_date DATE NOT NULL,
    status VARCHAR(20) NOT NULL,
    net_amount MONEY NOT NULL,
    tax MONEY NOT NULL,
    total_amount MONEY NOT NULL,
    CONSTRAINT fk_customer_order FOREIGN KEY (customer_id) REFERENCES CUSTOMERS (customer_id),
    CONSTRAINT fk_payment_order FOREIGN KEY (payment_id) REFERENCES PAYMENTS (payment_id)
);

-- Bảng ORDER_DETAILS
CREATE TABLE ORDER_DETAILS (
    od_id SERIAL PRIMARY KEY,
    order_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    quantity INT NOT NULL,
    CONSTRAINT fk_order_details_order FOREIGN KEY (order_id) REFERENCES ORDERS (order_id),
    CONSTRAINT fk_order_details_product FOREIGN KEY (product_id) REFERENCES PRODUCTS (product_id)
);
